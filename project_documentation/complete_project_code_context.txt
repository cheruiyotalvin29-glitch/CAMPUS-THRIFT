// ============================================================================
// STUDENT AUTHENTICATION & SESSION MANAGEMENT MODULE
// Strictly connected to Supabase Auth & PostgreSQL Profiles (No Stubs)
// ============================================================================

import { supabase, isSupabaseConfigured, logSupabase } from './supabaseClient.js';

// Module internal state
let currentUser = null;
let currentProfile = null;
let authChangeCallbacks = [];

/**
 * Initializes the Supabase authentication listener and retrieves any active session.
 */
export async function initAuth() {
    logSupabase('AUTH', 'Initializing authentication session...');

    if (!isSupabaseConfigured()) {
        logSupabase('AUTH', 'Cannot initialize auth: Supabase is not configured yet');
        currentUser = null;
        currentProfile = null;
        updateAuthUI();
        notifyAuthSubscribers();
        return null;
    }

    try {
        const { data: { session }, error } = await supabase.auth.getSession();
        
        if (error) {
            logSupabase('ERROR', 'Error fetching active Supabase session', error);
            throw error;
        }

        if (session && session.user) {
            currentUser = session.user;
            logSupabase('AUTH', 'Active student session restored', {
                id: currentUser.id,
                email: currentUser.email,
                role: currentUser.role
            });
            setTimeout(() => {
                loadUserProfile(currentUser.id);
            }, 0);
        } else {
            currentUser = null;
            currentProfile = null;
            logSupabase('AUTH', 'No active session found (student is logged out)');
        }

        // Attach real-time auth state listener
        supabase.auth.onAuthStateChange((event, session) => {
            logSupabase('AUTH', `Auth state changed: ${event}`, {
                hasSession: Boolean(session),
                userId: session?.user?.id
            });

            currentUser = session ? session.user : null;
            if (!currentUser) {
                currentProfile = null;
            }

            notifyAuthSubscribers();
            updateAuthUI();

            if (currentUser) {
                // Defer profile fetch out of auth state change handler to avoid lock contention
                setTimeout(async () => {
                    await loadUserProfile(currentUser.id);
                    notifyAuthSubscribers();
                    updateAuthUI();
                }, 0);
            }
        });
    } catch (err) {
        logSupabase('ERROR', 'Auth initialization error', { message: err.message });
    }

    updateAuthUI();
    notifyAuthSubscribers();
    return currentUser;
}

/**
 * Registers an observer callback for authentication state changes.
 * @param {Function} cb 
 */
export function onAuthStateChanged(cb) {
    if (typeof cb === 'function') {
        authChangeCallbacks.push(cb);
        cb(currentUser, currentProfile);
    }
}

function notifyAuthSubscribers() {
    authChangeCallbacks.forEach(cb => {
        try {
            cb(currentUser, currentProfile);
        } catch (e) {
            console.error('Subscriber callback error:', e);
        }
    });
}

/**
 * Returns current authenticated Supabase user or null.
 * @returns {object|null}
 */
export function getCurrentUser() {
    return currentUser;
}

/**
 * Returns current student profile row or null.
 * @returns {object|null}
 */
export function getCurrentProfile() {
    return currentProfile;
}

/**
 * Loads seller profile from the `profiles` table in Supabase.
 * @param {string} userId 
 */
export async function loadUserProfile(userId) {
    if (!userId || !isSupabaseConfigured()) return null;

    try {
        logSupabase('DATABASE', 'Fetching profile row from public.profiles', { userId });
        const { data, error } = await supabase
            .from('profiles')
            .select('*')
            .eq('id', userId)
            .single();

        if (error) {
            logSupabase('ERROR', 'Failed to retrieve profile row', error);
            return null;
        }

        currentProfile = data;
        logSupabase('DATABASE', 'Profile retrieved successfully', currentProfile);
        updateAuthUI();
        return currentProfile;
    } catch (e) {
        logSupabase('ERROR', 'Exception during profile load', { message: e.message });
        return null;
    }
}

/**
 * Signs up a new student account strictly through Supabase Auth.
 * @param {string} email 
 * @param {string} password 
 * @param {string} fullName 
 * @param {string} phoneNumber 
 * @param {string} campusLocation 
 */
export async function signUpStudent(email, password, fullName, phoneNumber = '', campusLocation = 'Main Campus') {
    if (!isSupabaseConfigured()) {
        throw new Error('Supabase is not configured. Please enter your project URL and Anon key first.');
    }

    if (!email || !password) {
        throw new Error('Please enter both student email and password.');
    }

    const cleanEmail = email.trim().toLowerCase();

    // 1. Proactively check if the email already exists in the database
    try {
        const { data: alreadyExists, error: rpcError } = await supabase.rpc('check_email_exists', {
            email_to_check: cleanEmail
        });
        if (alreadyExists) {
            logSupabase('AUTH', 'Sign-up blocked: email already registered in Supabase', { email: cleanEmail });
            throw new Error('Email already exists. Please sign in instead.');
        }
    } catch (err) {
        if (err.message && err.message.includes('already exists')) {
            throw err;
        }
        // Fallback: proceed to Supabase auth signup
    }

    logSupabase('AUTH', 'Initiating Supabase student registration', { email: cleanEmail, fullName, campusLocation });

    const { data, error } = await supabase.auth.signUp({
        email: cleanEmail,
        password,
        options: {
            data: {
                full_name: fullName,
                phone_number: phoneNumber,
                campus_location: campusLocation
            }
        }
    });

    if (error) {
        logSupabase('ERROR', 'Supabase registration failed', error);
        if (error.message && error.message.toLowerCase().includes('rate limit')) {
            throw new Error('Supabase email rate limit reached. Fix: In your Supabase Dashboard -> Authentication -> Providers -> Email, turn off "Confirm email".');
        }
        if (error.message && (
            error.message.toLowerCase().includes('already registered') || 
            error.message.toLowerCase().includes('already exists') || 
            error.message.toLowerCase().includes('unique') ||
            error.status === 400 || 
            error.status === 422 || 
            error.code === 'user_already_exists'
        )) {
            throw new Error('Email already exists. Please sign in instead.');
        }
        throw error;
    }

    // 2. Check for Supabase Email Enumeration Protection:
    // When a duplicate email is signed up and protection is enabled, Supabase returns data.user with an empty identities array!
    if (data.user && Array.isArray(data.user.identities) && data.user.identities.length === 0) {
        logSupabase('AUTH', 'Supabase returned empty identities (user already registered)', { email: cleanEmail });
        throw new Error('Email already exists. Please sign in instead.');
    }

    logSupabase('AUTH', 'Supabase user registration created', {
        userId: data.user?.id,
        email: data.user?.email,
        confirmationSent: !data.session
    });

    if (data.user) {
        currentUser = data.user;
        
        // Ensure profile row exists in public.profiles
        try {
            logSupabase('DATABASE', 'Upserting profile row in public.profiles', {
                id: data.user.id,
                full_name: fullName,
                phone_number: phoneNumber,
                campus_location: campusLocation
            });

            const { error: profileError } = await supabase
                .from('profiles')
                .upsert({
                    id: data.user.id,
                    full_name: fullName || 'Campus Student',
                    phone_number: phoneNumber || '',
                    campus_location: campusLocation || 'Main Campus'
                });

            if (profileError) {
                logSupabase('ERROR', 'Profile upsert returned error', profileError);
            } else {
                logSupabase('DATABASE', 'Profile upsert succeeded');
            }
        } catch (err) {
            logSupabase('ERROR', 'Profile creation exception', { message: err.message });
        }

        await loadUserProfile(currentUser.id);
    }

    updateAuthUI();
    notifyAuthSubscribers();
    return data;
}

/**
 * Authenticates an existing student strictly through Supabase Auth.
 * @param {string} email 
 * @param {string} password 
 */
export async function signInStudent(email, password) {
    if (!isSupabaseConfigured()) {
        throw new Error('Supabase is not configured. Please enter your project URL and Anon key first.');
    }

    if (!email || !password) {
        throw new Error('Please enter both email and password.');
    }

    logSupabase('AUTH', 'Attempting student sign-in with password', { email });

    const { data, error } = await supabase.auth.signInWithPassword({
        email,
        password
    });

    if (error) {
        logSupabase('ERROR', 'Supabase sign-in failed', error);
        throw error;
    }

    currentUser = data.user;

/**
 * Retrieves marketplace listings strictly from the Supabase `items` table.
 */
export async function fetchItems() {
    const spinner = document.getElementById('loading-spinner');
    if (spinner) spinner.classList.remove('hidden');

    if (!isSupabaseConfigured()) {
        logSupabase('DATABASE', 'Supabase credentials not configured. Skipping remote fetch.');
        if (spinner) spinner.classList.add('hidden');
        allItems = [];
        updateStats(0);
        renderItems([]);
        showUnconfiguredNotice();
        return;
    }

    try {
        logSupabase('DATABASE', 'Querying public.items (SELECT * ORDER BY created_at DESC)...');
        const startTime = performance.now();

        const { data, error } = await supabase
            .from('items')
            .select('*')
            .order('created_at', { ascending: false });

        const duration = Math.round(performance.now() - startTime);

        if (error) {
            logSupabase('ERROR', `Supabase query failed after ${duration}ms`, error);
            showToast('Database error: ' + error.message, 'error');
            allItems = [];
        } else {
            allItems = data || [];
            logSupabase('DATABASE', `Successfully fetched ${allItems.length} listings in ${duration}ms`, {
                itemCount: allItems.length,
                records: allItems
            });
        }
    } catch (err) {
        logSupabase('ERROR', 'Unexpected exception during fetchItems', { message: err.message });
        allItems = [];
    } finally {
        if (spinner) spinner.classList.add('hidden');
        updateStats();
        filterItems();
    }
}

export function filterItems() {
    const searchInput = document.getElementById('search-input');
    const query = (searchInput?.value || '').toLowerCase().trim();
    const clearBtn = document.getElementById('clear-search-btn');
    const sortFilter = document.getElementById('sort-filter')?.value || 'newest';
    const currentUser = getCurrentUser();

    if (clearBtn) {
        clearBtn.classList.toggle('hidden', query.length === 0);
    }

    let filtered = allItems.filter(item => {
        const titleMatch = (item.title || '').toLowerCase().includes(query);
        const descMatch = (item.description || '').toLowerCase().includes(query);
        const locationMatch = (item.campus_location || '').toLowerCase().includes(query);
        const matchesQuery = !query || titleMatch || descMatch || locationMatch;

        const matchesCategory = (activeCategory === 'ALL') || (item.category === activeCategory);

        return matchesQuery && matchesCategory;
    });

    if (sortFilter === 'price-asc') {
        filtered.sort((a, b) => Number(a.price) - Number(b.price));
    } else if (sortFilter === 'price-desc') {
        filtered.sort((a, b) => Number(b.price) - Number(a.price));
    } else {
        filtered.sort((a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0));
    }

    renderItems(filtered);
    updateStats(filtered.length);
}

// 3. CONTACT URI GENERATION (WhatsApp, Email, Phone)
// ----------------------------------------------------------------------------

export function buildContactURI(method, value, title) {
    const safeTitle = title || 'your campus thrift listing';
    const encodedTitle = encodeURIComponent(safeTitle);

    if (method === 'WhatsApp') {
        const cleanNumber = (value || '').replace(/[^0-9]/g, '');
        const message = encodeURIComponent(`Hi! I'm a student interested in your campus thrift listing: "${safeTitle}". Is it still available?`);
        return `https://wa.me/${cleanNumber}?text=${message}`;
    } else if (method === 'Email') {
        const subject = encodeURIComponent(`Campus Thrift Inquiry: ${safeTitle}`);
        const body = encodeURIComponent(`Hi,\n\nI saw your listing for "${safeTitle}" on Campus Thrift Market and would like to arrange purchase and campus handover.\n\nThanks!`);
        return `mailto:${value}?subject=${subject}&body=${body}`;
    } else {
        const cleanPhone = (value || '').trim();
        return `tel:${cleanPhone}`;
    }
}

        submitBtn.disabled = true;
        submitBtn.innerHTML = `
            <svg class="animate-spin -ml-1 mr-2 h-4 w-4 text-white inline" fill="none" viewBox="0 0 24 24">
                <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
                <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8v8H4z"></path>
            </svg>
            ${editingItemId ? 'Saving Changes...' : 'Uploading to Supabase...'}
        `;
    }

    const title = document.getElementById('item-title').value.trim();
    const price = parseFloat(document.getElementById('item-price').value);
    const category = document.getElementById('item-category').value;
    const condition = document.getElementById('item-condition').value;
    const location = document.getElementById('item-location').value.trim() || 'Main Campus';
    const description = document.getElementById('item-description').value.trim();
    const contactMethod = document.getElementById('item-contact-method').value;
    const contactValue = document.getElementById('item-contact-value').value.trim();
    const imageInput = document.getElementById('item-image');
    const imageFile = imageInput && imageInput.files ? imageInput.files[0] : null;

    let imageUrl = null;

    try {
        // Step 1: Upload Image to Supabase Cloud Storage bucket 'item-images' if a new file was chosen
        if (imageFile) {
            const fileExt = imageFile.name.split('.').pop();
            const fileName = `${Date.now()}-${Math.random().toString(36).substring(2, 7)}.${fileExt}`;
            const filePath = `items/${fileName}`;

            logSupabase('STORAGE', `Uploading item image to Supabase Storage bucket 'item-images'`, {
                filePath,
                fileSize: imageFile.size,
                fileType: imageFile.type
            });

            const { data: uploadData, error: uploadError } = await supabase.storage
                .from('item-images')
                .upload(filePath, imageFile, {
                    cacheControl: '3600',
                    upsert: false
                });

            if (uploadError) {
                logSupabase('ERROR', 'Supabase image upload failed', uploadError);
                throw new Error(`Storage upload failed: ${uploadError.message}. Make sure 'item-images' bucket is created.`);
            }

            const { data: urlData } = supabase.storage.from('item-images').getPublicUrl(filePath);
            imageUrl = urlData.publicUrl;
            logSupabase('STORAGE', 'Image uploaded successfully. Public URL generated', { imageUrl });
        }

        if (editingItemId) {
            // Step 2A: Update existing row in Supabase PostgreSQL 'items' table
            const updatePayload = {
                title,
                price,
                category,
                condition,
                campus_location: location,
                description,
                contact_method: contactMethod,
                contact_value: contactValue
            };
            if (imageUrl) {
                updatePayload.image_url = imageUrl;
            }

            logSupabase('DATABASE', `Updating item ${editingItemId} in public.items`, updatePayload);

            const { data: updatedData, error: updateError } = await supabase
                .from('items')
                .update(updatePayload)
                .eq('id', editingItemId)
                .eq('user_id', currentUser.id)
                .select()
                .single();

            if (updateError) {
                logSupabase('ERROR', 'Supabase item update failed', updateError);
                throw updateError;
            }

            logSupabase('DATABASE', 'Item successfully updated in Supabase items table', updatedData);
            showToast('✅ Listing updated successfully!', 'success');
            editingItemId = null;
        } else {
            // Step 2B: Insert new row into Supabase PostgreSQL 'items' table
            const itemRecord = {
                user_id: currentUser.id,
                title,
                price,
                category,
                condition,
                campus_location: location,
                description,
                image_url: imageUrl,
                contact_method: contactMethod,
                contact_value: contactValue
            };

            logSupabase('DATABASE', 'Inserting item row into public.items', itemRecord);

            const { data: insertedData, error: insertError } = await supabase
                .from('items')
                .insert([itemRecord])
                .select()
                .single();

            if (insertError) {
                logSupabase('ERROR', 'Supabase item insertion failed', insertError);
                throw insertError;
            }

            logSupabase('DATABASE', 'Item successfully persisted in Supabase items table', insertedData);
            showToast('🎉 Listing published live to Supabase!', 'success');
        }

        document.getElementById('post-item-form').reset();
        removeImagePreview();
        toggleModal('post-modal', false);
        await fetchItems();
        renderMyListings();
    } catch (err) {
        logSupabase('ERROR', 'Error during item creation/update pipeline', { message: err.message });
        showToast('Error saving listing: ' + err.message, 'error');
    } finally {
        if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.innerHTML = `<span id="submit-item-btn-text">${editingItemId ? 'Save Changes' : 'Publish Listing'}</span>`;
        }
    }
}

export async function executeDeleteConfirmedItem() {
    if (!pendingDeleteItemId) return;
    const itemId = pendingDeleteItemId;

    const confirmBtn = document.getElementById('confirm-delete-action-btn');
    if (confirmBtn) {
        confirmBtn.disabled = true;
        confirmBtn.innerHTML = `
            <svg class="animate-spin -ml-1 mr-2 h-4 w-4 text-white inline" fill="none" viewBox="0 0 24 24">
                <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
                <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8v8H4z"></path>
            </svg>
            Deleting...
        `;
    }

    try {
        logSupabase('DATABASE', `Executing DELETE FROM public.items WHERE id = '${itemId}'`);
        const { error } = await supabase
            .from('items')
            .delete()
            .eq('id', itemId);

        if (error) {
            logSupabase('ERROR', 'Supabase item deletion failed', error);
            throw error;
        }

        logSupabase('DATABASE', `Item ${itemId} deleted successfully from Supabase`);
        showToast('Listing removed permanently.', 'info');

        closeDeleteConfirmModal();
        toggleModal('detail-modal', false);

        await fetchItems();
        renderMyListings();
    } catch (err) {
        showToast('Could not delete listing: ' + err.message, 'error');
        if (confirmBtn) {
            confirmBtn.disabled = false;
            confirmBtn.innerHTML = `<span>Delete</span>`;
        }
    }
}

export async function handleAuth(event) {
    event.preventDefault();
    const email = document.getElementById('auth-email').value.trim();
    const password = document.getElementById('auth-password').value;
    const name = document.getElementById('auth-name').value.trim();
    const phone = document.getElementById('auth-phone').value.trim();
    const submitBtn = document.getElementById('auth-submit-btn');

    if (submitBtn) {
        submitBtn.disabled = true;
        submitBtn.innerText = isSignUpMode ? 'Registering with Supabase...' : 'Signing in with Supabase...';
    }

    try {
        if (isSignUpMode) {
            await signUpStudent(email, password, name, phone);
            showToast('Student account registered in Supabase! You can now post listings.', 'success');
        } else {
            await signInStudent(email, password);
            showToast('Welcome back! Signed in to Supabase successfully.', 'success');
        }
        toggleModal('auth-modal', false);
        await fetchItems();
    } catch (err) {
        showToast(err.message, 'error');
        // If email already exists during sign up, switch to Sign In mode for seamless experience
        if (isSignUpMode && err.message && err.message.toLowerCase().includes('already exists')) {
            toggleAuthMode();
            const emailInput = document.getElementById('auth-email');
            const pwdInput = document.getElementById('auth-password');
            if (emailInput) emailInput.value = email;
            if (pwdInput) {
                pwdInput.value = '';
                pwdInput.focus();
            }
        }
    } finally {
        if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.innerText = isSignUpMode ? 'Create Account' : 'Sign In';
        }
    }
}

export function toggleAuthMode() {
    isSignUpMode = !isSignUpMode;
    const title = document.getElementById('auth-title');
    const subtitle = document.getElementById('auth-subtitle');
    const submitBtn = document.getElementById('auth-submit-btn');
    const signupFields = document.getElementById('signup-fields');
    const toggleText = document.getElementById('auth-toggle-text');
    const toggleBtn = document.getElementById('toggle-auth-btn');

    if (isSignUpMode) {
        if (title) title.innerText = 'Student Registration';
        if (subtitle) subtitle.innerText = 'Create your free account to sell items on campus';
        if (submitBtn) submitBtn.innerText = 'Create Account';
        if (signupFields) signupFields.classList.remove('hidden');
        if (toggleText) toggleText.innerText = 'Already have an account?';
        if (toggleBtn) toggleBtn.innerText = 'Sign In';
    } else {
        if (title) title.innerText = 'Student Sign In';
        if (subtitle) subtitle.innerText = 'Sign in with your student credentials to post items';
        if (submitBtn) submitBtn.innerText = 'Sign In';
        if (signupFields) signupFields.classList.add('hidden');
        if (toggleText) toggleText.innerText = 'Need a student account?';
        if (toggleBtn) toggleBtn.innerText = 'Create Account';
    }
}

    if (titleInput) titleInput.value = item.title || '';
    if (priceInput) priceInput.value = item.price || '';
    if (categoryInput) categoryInput.value = item.category || 'Other';
    if (conditionInput) conditionInput.value = item.condition || 'Gently Used';
    if (locationInput) locationInput.value = item.campus_location || '';
    if (descriptionInput) descriptionInput.value = item.description || '';
    if (contactMethodInput) contactMethodInput.value = item.contact_method || 'WhatsApp';
    if (contactValueInput) contactValueInput.value = item.contact_value || '';

    // Image preview handling
    const previewContainer = document.getElementById('image-preview-container');
    const previewImageEl = document.getElementById('image-preview');
    const uploadPrompt = document.getElementById('upload-prompt');
    const imageInput = document.getElementById('item-image');
    if (imageInput) imageInput.value = '';

    if (item.image_url && previewContainer && previewImageEl && uploadPrompt) {
        previewImageEl.src = item.image_url;
        previewContainer.classList.remove('hidden');
        uploadPrompt.classList.add('hidden');
    } else {
        removeImagePreview();
    }

    // Update modal headings and submit button text
    const titleEl = document.getElementById('post-modal-title');
    const subtitleEl = document.getElementById('post-modal-subtitle');
    const btnTextEl = document.getElementById('submit-item-btn-text');

    if (titleEl) titleEl.textContent = 'Edit Campus Listing';
    if (subtitleEl) subtitleEl.textContent = 'Update price, location, condition, description, or contact information.';
    if (btnTextEl) btnTextEl.textContent = 'Save Changes';

    // Close other overlays and open post modal in edit mode
    toggleModal('detail-modal', false);
    toggleModal('my-listings-modal', false);
    toggleModal('post-modal', true);
}

export function previewImage(input) {
    if (input.files && input.files[0]) {
        const file = input.files[0];
        if (file.size > 5 * 1024 * 1024) {
            showToast('Image file size must be less than 5MB.', 'error');
            input.value = '';
            return;
        }

        const reader = new FileReader();
        reader.onload = function(e) {
            const preview = document.getElementById('image-preview');
            const previewContainer = document.getElementById('image-preview-container');
            const uploadPrompt = document.getElementById('upload-prompt');
            
            if (preview) preview.src = e.target.result;
            if (previewContainer) previewContainer.classList.remove('hidden');
            if (uploadPrompt) uploadPrompt.classList.add('hidden');
        };
        reader.readAsDataURL(file);
    }
}

export function removeImagePreview(event) {

/**
 * Automatically copies phone or WhatsApp contact number to clipboard.
 * @param {string} text 
 * @param {HTMLElement} btnElement 
 */
export async function copyToClipboard(text, btnElement) {
    if (!text) return;
    try {
        await navigator.clipboard.writeText(text);
        showToast(`📋 Copied "${text}" to clipboard!`, 'success');
        
        if (btnElement) {
            const originalHTML = btnElement.innerHTML;
            btnElement.classList.add('bg-emerald-600');
            btnElement.classList.remove('bg-indigo-600');
            btnElement.innerHTML = `<span>✓</span><span>Copied!</span>`;
            setTimeout(() => {
                btnElement.innerHTML = originalHTML;
                btnElement.classList.remove('bg-emerald-600');
                btnElement.classList.add('bg-indigo-600');
            }, 2000);
        }
    } catch (err) {
        const textArea = document.createElement('textarea');
        textArea.value = text;
        textArea.style.position = 'fixed';
        textArea.style.opacity = '0';
        document.body.appendChild(textArea);
        textArea.focus();
        textArea.select();
        try {
            document.execCommand('copy');
            showToast(`📋 Copied "${text}" to clipboard!`, 'success');
        } catch (e) {
            showToast(`Contact: ${text}`, 'info');
        }
        document.body.removeChild(textArea);
    }
}

export function showToast(message, type = 'info') {
    const container = document.getElementById('toast-container');
    if (!container) return;

    const toast = document.createElement('div');
    toast.className = `toast toast-${type}`;

    let icon = 'ℹ️';
    if (type === 'success') icon = '✅';
    if (type === 'error') icon = '⚠️';

    toast.innerHTML = `
        <span class="text-base flex-shrink-0">${icon}</span>
        <span class="flex-grow text-xs font-medium text-slate-800">${escapeHTML(message)}</span>
    `;

    container.appendChild(toast);

    setTimeout(() => {
        toast.style.opacity = '0';
        toast.style.transform = 'translateY(-10px)';
        setTimeout(() => toast.remove(), 300);
    }, 4000);
}

function escapeHTML(str) {
    if (!str) return '';
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

);

CREATE TABLE IF NOT EXISTS public.items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
    title TEXT NOT NULL,
    description TEXT NOT NULL,
    price NUMERIC NOT NULL CHECK (price >= 0),
    category TEXT NOT NULL CHECK (category IN ('Textbooks', 'Electronics', 'Clothing', 'Dorm Essentials', 'Other')),
    condition TEXT DEFAULT 'Gently Used' CHECK (condition IN ('Brand New', 'Like New', 'Gently Used', 'Fair')),
    campus_location TEXT DEFAULT 'Main Campus',
    image_url TEXT,
    contact_method TEXT NOT NULL CHECK (contact_method IN ('WhatsApp', 'Phone')),
    contact_value TEXT NOT NULL, -- Target phone number or WhatsApp number
    is_sold BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Enable Row Level Security (RLS) on All Tables
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.items ENABLE ROW LEVEL SECURITY;

-- 5. Row Level Security Policies for `profiles`
DROP POLICY IF EXISTS "Public profiles are viewable by everyone" ON public.profiles;
DROP POLICY IF EXISTS "Users can insert their own profile" ON public.profiles;
DROP POLICY IF EXISTS "Users can update their own profile" ON public.profiles;

-- Allow public read access to seller profiles
CREATE POLICY "Public profiles are viewable by everyone" 
ON public.profiles FOR SELECT 
USING (true);

-- Allow authenticated users to insert their own profile record
CREATE POLICY "Users can insert their own profile" 
ON public.profiles FOR INSERT 
WITH CHECK (auth.uid() = id);

-- Allow users to modify only their own profile record
CREATE POLICY "Users can update their own profile" 
ON public.profiles FOR UPDATE 
USING (auth.uid() = id);

DROP POLICY IF EXISTS "Authenticated users can create items" ON public.items;
DROP POLICY IF EXISTS "Users can update their own items" ON public.items;

ON public.items FOR SELECT 
USING (true);

-- Allow logged-in students to create new listings tied strictly to their auth ID
CREATE POLICY "Authenticated users can create items" 
ON public.items FOR INSERT 
WITH CHECK (
    auth.role() = 'authenticated' 
    AND auth.uid() = user_id
);

-- Allow sellers to modify only their own active listings
CREATE POLICY "Users can update their own items" 
ON public.items FOR UPDATE 
USING (auth.uid() = user_id);

-- Allow sellers to delete only their own listings
CREATE POLICY "Users can delete their own items"

INSERT INTO storage.buckets (id, name, public)

DROP POLICY IF EXISTS "Authenticated users can upload item images" ON storage.objects;

USING (
    bucket_id = 'item-images'

-- 8. Automated Trigger: Automatically Create Profile on User Sign Up (Security Definer)
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.profiles (id, full_name, phone_number, campus_location)
    VALUES (
        new.id,
        COALESCE(new.raw_user_meta_data->>'full_name', split_part(new.email, '@', 1)),
        COALESCE(new.raw_user_meta_data->>'phone_number', ''),
        COALESCE(new.raw_user_meta_data->>'campus_location', 'Main Campus')
    )
    ON CONFLICT (id) DO UPDATE SET
        full_name = COALESCE(EXCLUDED.full_name, public.profiles.full_name),
        phone_number = COALESCE(EXCLUDED.phone_number, public.profiles.phone_number),
        campus_location = COALESCE(EXCLUDED.campus_location, public.profiles.campus_location);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Recreate trigger
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
    AFTER INSERT ON auth.users
    FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- 9. Automated Trigger: Instant Email Confirmation (No Confirmation Required!)
CREATE OR REPLACE FUNCTION public.auto_confirm_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    NEW.email_confirmed_at = COALESCE(NEW.email_confirmed_at, NOW());
    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_auto_confirm ON auth.users;
CREATE TRIGGER on_auth_user_auto_confirm
    BEFORE INSERT ON auth.users
    FOR EACH ROW EXECUTE FUNCTION public.auto_confirm_new_user();

UPDATE auth.users SET email_confirmed_at = NOW() WHERE email_confirmed_at IS NULL;

-- Safe migration for existing tables:
ALTER TABLE public.items ADD COLUMN IF NOT EXISTS campus_location TEXT DEFAULT 'Main Campus';
UPDATE public.items SET contact_method = 'WhatsApp' WHERE contact_method = 'Email';
ALTER TABLE public.items DROP CONSTRAINT IF EXISTS items_contact_method_check;
ALTER TABLE public.items ADD CONSTRAINT items_contact_method_check 
    CHECK (contact_method IN ('WhatsApp', 'Phone'));

-- 11. RPC Helper: Check If Email Already Exists (Prevents Duplicate Sign-Ups)
CREATE OR REPLACE FUNCTION public.check_email_exists(email_to_check TEXT)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, auth
AS $$
BEGIN

        SELECT 1 FROM auth.users 
        WHERE LOWER(email) = LOWER(TRIM(email_to_check))
    );
END;
$$;

import http.server
import socketserver
import sys

class RobustHTTPRequestHandler(http.server.SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header('Cache-Control', 'no-cache, no-store, must-revalidate')
        self.send_header('Pragma', 'no-cache')
        self.send_header('Expires', '0')
        self.send_header('Access-Control-Allow-Origin', '*')
        super().end_headers()

    def log_message(self, format, *args):
        sys.stderr.write(f"[{self.log_date_time_string()}] {format % args}\n")
        sys.stderr.flush()

class ThreadingServer(socketserver.ThreadingMixIn, http.server.HTTPServer):
    daemon_threads = True
    allow_reuse_address = True

if __name__ == '__main__':
    port = int(sys.argv[1]) if len(sys.argv) > 1 else 8000
    server_address = ('0.0.0.0', port)
    httpd = ThreadingServer(server_address, RobustHTTPRequestHandler)
    print(f"Robust HTTP Server running at http://127.0.0.1:{port} and http://localhost:{port}", flush=True)
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        pass

# 🎓 Campus Online Thrift Market

> A decentralized, low-complexity peer-to-peer (P2P) campus thrift marketplace designed for university students to buy, sell, and hand down textbooks, electronics, dorm essentials, and clothing with zero transaction fees and direct WhatsApp contact.

---

## 🌟 Architectural Overview

Traditional campus commerce on instant messaging groups and social media status posts suffers from rapid message depreciation, lack of persistent indexing, absence of category filtering, and high friction. 

**Campus Online Thrift Market** solves this with a **Zero-Build-Step Single-Page Application (SPA)** backed directly by **Supabase Backend-as-a-Service (BaaS)**:

- **Frontend:** Pure HTML5, Modern Vanilla JavaScript (ES6+ Modules), and utility-first Tailwind CSS loaded via CDN.
- **Backend:** Supabase PostgreSQL with automated Row Level Security (RLS) policies.
- **Authentication:** Supabase Auth (JWT session management, email/password verification).
- **File Storage:** Supabase Cloud Storage bucket (`item-images`) with public CDN URLs.
- **P2P Communication:** Direct deep-linked URI protocols (`https://wa.me/`, `mailto:`, `tel:`) bypassing server-side chat overhead.
- **Zero Node.js Rot:** No `npm install`, Webpack, Vite, or local transpilation pipelines required.

## ⚡ Supabase Cloud Backend Setup

To connect the application to your live Supabase cloud backend:

### Step 1: Create a Supabase Project
1. Log in to [Supabase](https://supabase.com/) and create a new project.
2. Note your **Project URL** and **Anon Public API Key** from **Project Settings > API**.

### Step 2: Run Database Initialization Script
1. Navigate to the **SQL Editor** tab in the Supabase Dashboard.
2. Open [`sql/schema.sql`](file:///C:/Users/G_R_E/Documents/REPOSITORIES/campus%20thrift%20market/sql/schema.sql) in this repository and paste the entire script into the SQL editor.
3. Click **Run**.
   - This sets up:
     - `public.profiles` (linked to `auth.users`)
     - `public.items` (with price constraints and contact details)
     - Row Level Security (RLS) policies for secure public reading and owner-only editing
     - Public storage bucket `item-images` with upload permissions
     - Automatic user profile trigger `on_auth_user_created`

### Step 3: Link Credentials
You can connect your credentials in **two ways**:

- **Option A (In-App GUI - Easiest):**
  Open the app in your browser, click the **Supabase Ready / Backend Config** pill in the top header, paste your `Project URL` and `Anon Key`, and click **Connect Supabase**.

- **Option B (Direct Code):**
  In [`js/supabaseClient.js`](file:///C:/Users/G_R_E/Documents/REPOSITORIES/campus%20thrift%20market/js/supabaseClient.js), replace the placeholder constants:
  ```javascript
  export const DEFAULT_SUPABASE_URL = 'https://YOUR_PROJECT_ID.supabase.co';
  export const DEFAULT_SUPABASE_ANON_KEY = 'YOUR_SUPABASE_ANON_KEY';
  ```
